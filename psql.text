--Задача 3: Система управления блогом с комментариями (FOREIGN KEY, TIMESTAMP)
create table Posts( --(Посты)
      id SERIAL primary key,
      title VARCHAR(200), -- название поста
      content TEXT, -- содержание поста
      author VARCHAR(100), --автор
      created_at TIMESTAMP, -- когда создан пост
      views INT --количество просмотров      
);
create table Comments(
       id SERIAL primary key,
       post_id INT, --ID поста (FOREIGN KEY на Posts!)
       author VARCHAR(100), --автор
       text TEXT, -- содержание поста
       created_at TIMESTAMP,
       CONSTRAINT fk_post
      foreign key (post_id)
      references posts(id)
      on delete cascade         
);

--Задача 5: Система управления школой (FOREIGN KEY ON DELETE CASCADE, JOIN)

create table Teachers (
  id SERIAL PRIMARY KEY,
  first_name VARCHAR(100),
  last_name VARCHAR(100),
  subject VARCHAR(50),
  salary DECIMAL(10, 2),
  hire_date DATE
);


create table Grades (
  id SERIAL PRIMARY key,
  student_name VARCHAR(100),
  teacher_id INT,
  subject VARCHAR(50),
  score INT,
  grade_date DATE,
  CONSTRAINT FK_teachers
  foreign key (teacher_id)
  references teachers (id)
  ON DELETE CASCADE
);
